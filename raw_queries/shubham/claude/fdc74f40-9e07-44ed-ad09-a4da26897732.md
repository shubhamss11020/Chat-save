## **Lock-Free SPSC Ring Buffer Implementation**

### **Key Design Principles:**

1. **Memory Ordering (Acquire/Release)**
   - `push()`: Uses `Ordering::Relaxed` to load producer index (local thread), then `Ordering::Acquire` to see consumer updates, and `Ordering::Release` when storing the advanced index
   - `pop()`: Mirror of push - `Ordering::Relaxed` for local reads, `Ordering::Acquire` for producer visibility
   - This ensures the buffer write happens-before the index advance becomes visible to the other thread

2. **False Sharing Prevention**
   - Producer and consumer indices are wrapped in `CacheAligned<T>` with `#[repr(align(64))]`
   - 64 bytes = typical x86_64/ARM64 cache line size
   - Prevents both indices from occupying the same cache line, eliminating RFO (Read-For-Ownership) ping-ponging

3. **Zero-Copy Post-Initialization**
   - Pre-allocated `Vec` becomes `Box<[Option<T>]>`
   - Power-of-2 capacity enables `& (capacity - 1)` modulo instead of `%`
   - No allocations during push/pop operations

Here's the complete implementation:

```rust
use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::Arc;

/// Align to cache line (64 bytes) to prevent false sharing
#[repr(align(64))]
struct CacheAligned<T>(T);

/// Lock-free SPSC ring buffer with Acquire/Release semantics
pub struct SpscRingBuffer<T> {
    buffer: Box<[Option<T>]>,
    capacity_mask: usize,
    // Producer index - only written by producer thread
    producer: Arc<CacheAligned<AtomicUsize>>,
    // Consumer index - only written by consumer thread
    consumer: Arc<CacheAligned<AtomicUsize>>,
}

impl<T> SpscRingBuffer<T> {
    /// Create a new ring buffer. Capacity must be power of 2.
    pub fn new(capacity: usize) -> Self {
        assert!(capacity > 0 && capacity.is_power_of_two(), 
                "capacity must be power of 2");
        let buffer = vec![None; capacity].into_boxed_slice();
        Self {
            buffer,
            capacity_mask: capacity - 1,
            producer: Arc::new(CacheAligned(AtomicUsize::new(0))),
            consumer: Arc::new(CacheAligned(AtomicUsize::new(0))),
        }
    }

    /// Push an element. Returns Err if buffer is full.
    /// Memory ordering: Release on index store ensures buffer write is visible.
    pub fn push(&mut self, value: T) -> Result<(), T> {
        let p_idx = self.producer.0.load(Ordering::Relaxed);
        let c_idx = self.consumer.0.load(Ordering::Acquire);
        let next = (p_idx + 1) & self.capacity_mask;
        
        // Full when next wraps to consumer's position
        if next == c_idx {
            return Err(value);
        }

        // Write value, then advance index with Release
        self.buffer[p_idx & self.capacity_mask] = Some(value);
        self.producer.0.store(next, Ordering::Release);
        Ok(())
    }

    /// Pop an element. Returns None if buffer is empty.
    /// Memory ordering: Acquire on index load ensures we see producer's writes.
    pub fn pop(&mut self) -> Option<T> {
        let c_idx = self.consumer.0.load(Ordering::Relaxed);
        let p_idx = self.producer.0.load(Ordering::Acquire);
        
        // Empty when indices match
        if c_idx == p_idx {
            return None;
        }

        // Take value, then advance index with Release
        let val = self.buffer[c_idx & self.capacity_mask].take();
        self.consumer.0.store((c_idx + 1) & self.capacity_mask, Ordering::Release);
        val
    }

    pub fn len(&self) -> usize {
        (self.producer.0.load(Ordering::Acquire)
            .wrapping_sub(self.consumer.0.load(Ordering::Acquire)))
            & self.capacity_mask
    }

    pub fn is_empty(&self) -> bool {
        self.producer.0.load(Ordering::Acquire) == self.consumer.0.load(Ordering::Acquire)
    }

    pub fn is_full(&self) -> bool {
        let p = self.producer.0.load(Ordering::Acquire);
        let c = self.consumer.0.load(Ordering::Acquire);
        ((p + 1) & self.capacity_mask) == c
    }

    pub fn capacity(&self) -> usize { self.buffer.len() }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_basic_operations() {
        let mut rb = SpscRingBuffer::new(8);
        for i in 0..5 { assert!(rb.push(i).is_ok()); }
        for i in 0..5 { assert_eq!(rb.pop(), Some(i)); }
        assert!(rb.is_empty());
    }

    #[test]
    fn test_wraparound() {
        let mut rb = SpscRingBuffer::new(4);
        for i in 0..4 { rb.push(i).ok(); }
        rb.pop(); rb.pop();
        rb.push(10).ok(); rb.push(11).ok();
        assert_eq!(rb.pop(), Some(2));
        assert_eq!(rb.pop(), Some(3));
    }

    #[test]
    fn test_concurrent() {
        use std::sync::{Arc, Barrier};
        use std::sync::atomic::AtomicBool;
        use std::thread;

        let rb = Arc::new(std::sync::Mutex::new(SpscRingBuffer::new(256)));
        let done = Arc::new(AtomicBool::new(false));
        let barrier = Arc::new(Barrier::new(2));
        let n = 100_000;

        let rb_p = Arc::clone(&rb);
        let b_p = Arc::clone(&barrier);
        let prod = thread::spawn(move || {
            b_p.wait();
            for i in 0..n {
                while rb_p.lock().unwrap().push(i).is_err() { thread::yield_now(); }
            }
        });

        let rb_c = Arc::clone(&rb);
        let d_c = Arc::clone(&done);
        let b_c = Arc::clone(&barrier);
        let cons = thread::spawn(move || {
            b_c.wait();
            let mut cnt = 0;
            loop {
                while rb_c.lock().unwrap().pop().is_some() { cnt += 1; }
                if d_c.load(Ordering::Acquire) && rb_c.lock().unwrap().is_empty() { break; }
            }
            cnt
        });

        prod.join().unwrap();
        done.store(true, Ordering::Release);
        let cnt = cons.join().unwrap();
        assert_eq!(cnt, n);
    }
}

    fn bench_multi_threaded_contention() {
        let rb = Arc::new(std::sync::Mutex::new(SpscRingBuffer::new(4096)));
        let done = Arc::new(AtomicBool::new(false));
        let barrier = Arc::new(Barrier::new(2));
        let n = 5_000_000;

        let rb_p = Arc::clone(&rb);
        let b_p = Arc::clone(&barrier);
        let prod = thread::spawn(move || {
            b_p.wait();
            let start = Instant::now();
            for i in 0..n {
                while rb_p.lock().unwrap().push(i).is_err() { thread::yield_now(); }
            }
            start.elapsed()
        });

        let rb_c = Arc::clone(&rb);
        let d_c = Arc::clone(&done);
        let b_c = Arc::clone(&barrier);
        let cons = thread::spawn(move || {
            b_c.wait();
            let start = Instant::now();
            let mut cnt = 0;
            loop {
                while rb_c.lock().unwrap().pop().is_some() { cnt += 1; }
                if d_c.load(Ordering::Acquire) && rb_c.lock().unwrap().is_empty() { break; }

        let avg = (t_p + t_c) / 2;
        println!("MT: {:.2}M ops/sec, {:.2}ns/op, count={}",
```
